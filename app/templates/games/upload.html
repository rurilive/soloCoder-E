3. Score Submission
// Platform injects these globals:
// window.GAME_SLUG - your game identifier
// window.submitGameScore(score) - submit score

// Example: Game over
async function gameOver(finalScore) {
    console.log("Score:", finalScore);
    
    // Submit to leaderboard
    const result = await window.submitGameScore(finalScore);
    if (result.success) {
        alert("Score submitted!");
    }
}

  • No external scripts - all code must be local
  • Use addEventListener() instead of inline handlers
  • No dangerous file types (.php, .exe, etc.)
  • No path traversal - no "../" in filenames
