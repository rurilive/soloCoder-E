4. Security Best Practices
  • No eval() - Never use eval() or Function constructor with user input
  • No inline event handlers - Use addEventListener() instead
  • Sanitize all user input - If your game accepts text input
  • No external scripts - All scripts must be included in your upload
  • CORS restrictions - Games run in sandboxed iframe, no cross-origin requests
5. Technical Limitations
  • Maximum file size: 10MB total
  • Allowed file types: .html, .js, .css, .json, .png, .jpg, .gif, .svg, .wav, .mp3, .ogg
  • No server-side code: Games are purely client-side
  • No database access: Use localStorage for game saves
  • iframe sandbox: Games run in isolated context
6. Example Game (Click Counter)
// js/game.js
let score = 0;
let gameActive = false;
let timeLeft = 10;
let timer = null;

const scoreEl = document.getElementById('score');
const timeEl = document.getElementById('time');
const clickBtn = document.getElementById('click-btn');
const startBtn = document.getElementById('start-btn');

startBtn.addEventListener('click', startGame);
clickBtn.addEventListener('click', () => {
    if (gameActive) {
        score++;
        scoreEl.textContent = score;
    }
});

function startGame() {
    gameActive = true;
    score = 0;
    timeLeft = 10;
    scoreEl.textContent = '0';
    timeEl.textContent = '10';
    startBtn.style.display = 'none';
    
    timer = setInterval(() => {
        timeLeft--;
        timeEl.textContent = timeLeft;
        if (timeLeft <= 0) {
            endGame();
        }
    }, 1000);
}

async function endGame() {
    gameActive = false;
    clearInterval(timer);
    startBtn.style.display = 'inline-block';
    
    // Submit score to platform
    await window.submitGameScore(score);
    alert(`Game Over! Your score: ${score}`);
}
